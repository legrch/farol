var idPattern = regexp.MustCompile(`^[A-Za-z0-9][A-Za-z0-9._:/-]{0,127}$`)
var utcPattern = regexp.MustCompile(`^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}(\.[0-9]{1,9})?Z$`)
var decimalPattern = regexp.MustCompile(`^(0|[1-9][0-9]{0,19})$`)

// Validate applies the one canonical closed schema. Framing also rejects duplicates.
func Validate(name string, raw []byte) error {
 if !ValidUnicodeJSON(raw) { return errors.New("invalid JSON") }
 var value interface{}
 decoder:=json.NewDecoder(strings.NewReader(string(raw))); decoder.UseNumber()
 if err:=decoder.Decode(&value); err!=nil { return err }
 if !valid(name,value,0) { return errors.New("invalid "+name) }
 return nil
}
func valid(kind string,value interface{},depth int) bool {
 if depth>32 { return false }
 if strings.HasSuffix(kind,"[]") && kind!="text[]" {a,ok:=value.([]interface{});if !ok || len(a)>128{return false};for _,item:=range a{if !valid(strings.TrimSuffix(kind,"[]"),item,depth+1){return false}};return true}
 switch kind {
 case "boolean": _,ok:=value.(bool);return ok
 case "nullableId": return value==nil || valid("id",value,depth+1)
 case "string": _,ok:=value.(string);return ok
 case "text","id","decimal","utc":
  s,ok:=value.(string); if !ok {return false}
  switch kind {
  case "id":return idPattern.MatchString(s)
  case "decimal":if !decimalPattern.MatchString(s){return false}; _,err:=strconv.ParseUint(s,10,64);return err==nil
  case "utc":if !utcPattern.MatchString(s){return false};t,err:=time.Parse(time.RFC3339Nano,s);return err==nil && strings.HasSuffix(s,"Z") && t.Year()>=1 && t.Year()<=9999
  default:return len(s)>0 && len(s)<=256 && !strings.ContainsAny(s,"\x00\r\n")
  }
 case "uint32","int32":
  n,ok:=value.(json.Number);if !ok{return false};i,err:=strconv.ParseFloat(string(n),64);if err!=nil || math.IsInf(i,0) || math.IsNaN(i) || math.Trunc(i)!=i{return false};if kind=="uint32"{return i>=0 && i<=math.MaxUint32};return i>=math.MinInt32 && i<=math.MaxInt32
 case "text[]":a,ok:=value.([]interface{});if !ok || len(a)>64{return false};for _,item:=range a{if !valid("text",item,depth+1){return false}};return true
 case "json":
  switch v:=value.(type){
  case nil,bool,string:return true
  case json.Number:n,err:=strconv.ParseFloat(string(v),64);return err==nil && !math.IsInf(n,0) && !math.IsNaN(n) && math.Abs(n)<=9007199254740991
  case []interface{}:for _,item:=range v{if !valid("json",item,depth+1){return false}};return true
  case map[string]interface{}:for _,item:=range v{if !valid("json",item,depth+1){return false}};return true
  default:return false
  }
 }
 def,known:=definitions[kind];if !known{return false};object,ok:=value.(map[string]interface{});if !ok{return false}
 for key:=range object{if _,known:=def.Fields[key];!known{return false}}
 for key,spec:=range def.Fields{
  item,present:=object[key];if !present{if !spec.Optional{return false};continue}
  if !valid(spec.Type,item,depth+1){return false}
  if spec.Minimum!=nil || spec.Maximum!=nil {n,ok:=item.(json.Number);if !ok{return false};i,err:=strconv.ParseFloat(string(n),64);if err!=nil || math.IsInf(i,0) || math.IsNaN(i) || math.Trunc(i)!=i || (spec.Minimum!=nil && i<float64(*spec.Minimum)) || (spec.Maximum!=nil && i>float64(*spec.Maximum)){return false}}
  if spec.MinBytes!=nil || spec.MaxBytes!=nil || spec.Pattern!="" {s,ok:=item.(string);if !ok{return false};if (spec.MinBytes!=nil && len(s)<*spec.MinBytes) || (spec.MaxBytes!=nil && len(s)>*spec.MaxBytes){return false};if spec.Pattern!="" {matched,err:=regexp.MatchString(spec.Pattern,s);if err!=nil || !matched{return false}}}
  if spec.MaxItems!=nil {a,ok:=item.([]interface{});if !ok || len(a)>*spec.MaxItems{return false}}
  if len(spec.Values)>0{matched:=false;for _,allowed:=range spec.Values{if item==allowed{matched=true}};if !matched{return false}}
 }
 if kind=="Response"{_,result:=object["result"];_,fault:=object["error"];if result==fault{return false};if result{if _,hasID:=object["id"];!hasID{return false}}}
 return true
}

// Typed decoding keeps integer JSON values interoperable with TypeScript,
// including equivalent decimal/exponent spellings, without changing opaque data.
func decodeIntegers(name string, raw []byte, target interface{}, fields []string) error {
 if err:=Validate(name,raw);err!=nil{return err}
 var object map[string]json.RawMessage
 if err:=json.Unmarshal(raw,&object);err!=nil{return err}
 for _,field:=range fields {
  number,present:=object[field];if !present{continue}
  value,err:=strconv.ParseFloat(string(number),64);if err!=nil{return err}
  object[field]=json.RawMessage(strconv.FormatInt(int64(value),10))
 }
 normalized,err:=json.Marshal(object);if err!=nil{return err}
 return json.Unmarshal(normalized,target)
}

// ValidUnicodeJSON rejects escaped lone surrogates before Go's JSON decoder can
// replace them with U+FFFD. This is shared by schema and framing validation.
func ValidUnicodeJSON(raw []byte) bool {
 if !utf8.Valid(raw) || !json.Valid(raw) { return false }
 quoted:=false
 for i:=0;i<len(raw);i++ {
  if raw[i]=='"' { quoted=!quoted;continue }
  if !quoted || raw[i]!='\\' { continue }
  if raw[i+1]!='u' { i++;continue }
  unit,_:=strconv.ParseUint(string(raw[i+2:i+6]),16,16)
  if unit>=0xdc00 && unit<=0xdfff { return false }
  if unit>=0xd800 && unit<=0xdbff {
   if i+12>len(raw) || raw[i+6]!='\\' || raw[i+7]!='u' { return false }
   low,err:=strconv.ParseUint(string(raw[i+8:i+12]),16,16)
   if err!=nil || low<0xdc00 || low>0xdfff { return false }
   i+=11
  } else { i+=5 }
 }
 return true
}
